HARTREE_TO_EV = 27.211386245988

def parse_orca(path):
    p = Path(path)
    text = p.read_text(encoding="utf-8", errors="replace") if p.exists() else ""
    energies = [float(m.group(1)) for m in re.finditer(r"FINAL SINGLE POINT ENERGY\s+([-+]?\d+(?:\.\d+)?)", text)]
    energy = energies[-1] if energies else None
    coords = []
    marker = "CARTESIAN COORDINATES (ANGSTROEM)"
    idx = text.rfind(marker)
    if idx != -1:
        lines = text[idx:].splitlines()
        start = None
        for i, line in enumerate(lines):
            if re.match(r"^\s*-{5,}", line):
                start = i + 1
                break
        if start is not None:
            for line in lines[start:]:
                if not line.strip():
                    break
                parts = line.split()
                if len(parts) >= 4 and re.fullmatch(r"[A-Za-z]{1,3}", parts[0]):
                    try:
                        coords.append([parts[0], float(parts[1]), float(parts[2]), float(parts[3])])
                    except ValueError:
                        pass
    return energy, coords


def write_xyz(path, coords):
    lines = [str(len(coords)), "archived ORCA final geometry; not optimized for this report"]
    for sym, x, y, z in coords:
        lines.append(f"{sym} {x:.8f} {y:.8f} {z:.8f}")
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")


records = []
for cid, path, original in CANDIDATES:
    energy, coords = parse_orca(path)
    if energy is not None:
        records.append({"candidate_id": cid, "path": path, "original_filename": original, "energy_hartree": energy, "coords": coords})

records.sort(key=lambda r: r["energy_hartree"])
min_energy = records[0]["energy_hartree"] if records else None
for r in records:
    r["relative_energy_kj_mol"] = (r["energy_hartree"] - min_energy) * HARTREE_TO_KJ if min_energy is not None else None
    r["relative_energy_ev"] = (r["energy_hartree"] - min_energy) * HARTREE_TO_EV if min_energy is not None else None
