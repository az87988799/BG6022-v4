"""Package existing, hash-verified hexane evidence; never execute task code.

Wait for the final continuation and its independent audit, then pass --finalize --audit
PATH. Only selected text files are copied verbatim; GBW and other binaries stay
outside the ZIP. The manifest reports recorded facts, not scientific approval.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
from datetime import datetime, timezone
from pathlib import Path, PurePosixPath
from zipfile import ZIP_DEFLATED, ZipFile, ZipInfo

RUN_ID = "run_4a5877907dbd4c7fbcfae9bdbd8f608e"
TABLES = ("candidate_structures.csv", "candidate_structures.json",
          "lowest_candidate_energy_ranking.csv", "lowest_candidate_energy_ranking.json",
          "failed_members.csv", "failed_members.json", "summary.json")
CANDIDATES = tuple(f"candidate_{number:02}.xyz" for number in range(1, 13))
MAX_TOTAL_BYTES = 10 * 1024 * 1024
MAX_FILE_BYTES = MAX_TOTAL_BYTES


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def encode(value):
    return (json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + "\n").encode("utf-8")


def read(path):
    raw = path.read_bytes()
    return json.loads(raw), raw


def build(batch, audit_path, output):
    store = batch.parent / "code-as-tools-step5/live-store"
    run_path = store / "runs" / RUN_ID / "run.json"
    run, run_raw = read(run_path)
    phases = sorted(path for path in batch.iterdir()
                    if path.is_dir() and re.fullmatch(r"continuation-[0-9]{3}", path.name))
    if not phases or [path.name for path in phases] != [f"continuation-{number:03}"
                                                       for number in range(1, len(phases) + 1)]:
        raise RuntimeError("Expected an uninterrupted continuation chain")
    outcome_path = phases[-1] / "outcome.json"
    outcome, outcome_raw = read(outcome_path)
    if run["id"] != RUN_ID or outcome["usage"] != run["usage"] or outcome["state"] != run["state"]:
        raise RuntimeError("Wait for the stable latest continuation outcome")
    if run["state"] in {"ready", "running"}:
        raise RuntimeError("A running Run is not a final delivery")
    audit, audit_raw = read(audit_path)
    audit_ids = [value for value in (audit.get("run_id"), audit.get("facts", {}).get("run_id"))
                 if value is not None]
    if not audit_ids or any(value != RUN_ID for value in audit_ids):
        raise RuntimeError("Independent audit must identify this same Run")
    artifact_cache, selected, payloads = {}, [], {}

    def artifact(identifier, expected=None):
        if not isinstance(identifier, str) or not re.fullmatch(r"[A-Za-z0-9_-]{1,96}", identifier):
            raise RuntimeError("Invalid Artifact identifier")
        if identifier not in artifact_cache:
            metadata_path = store / "artifacts" / identifier / "artifact.json"
            metadata, metadata_raw = read(metadata_path)
            path = (store / metadata["path"]).resolve()
            if metadata["id"] != identifier or not path.is_relative_to(store.resolve()):
                raise RuntimeError("Artifact identity or path mismatch")
            if path.stat().st_size > MAX_FILE_BYTES:
                raise RuntimeError("Selected text exceeds package per-file limit")
            raw = path.read_bytes()
            if sha(raw) != metadata["sha256"] or len(raw) != metadata["size"]:
                raise RuntimeError("Artifact hash or size changed: " + identifier)
            raw.decode("utf-8")
            if b"\x00" in raw:
                raise RuntimeError("Binary Artifact cannot enter this text delivery")
            artifact_cache[identifier] = metadata, raw, path, sha(metadata_raw)
        item = artifact_cache[identifier]
        if expected is not None and item[0]["sha256"] != expected:
            raise RuntimeError("Frozen reference differs from Artifact hash")
        return item

    def add(name, identifier, *, expected=None, purpose):
        if PurePosixPath(name).is_absolute() or ".." in PurePosixPath(name).parts or name in payloads:
            raise RuntimeError("Unsafe or duplicate package name")
        metadata, raw, _, metadata_hash = artifact(identifier, expected)
        payloads[name] = raw
        selected.append({"path": name, "purpose": purpose, "artifact_id": identifier,
            "sha256": metadata["sha256"], "bytes": len(raw), "role": metadata["role"],
            "attempt_id": metadata.get("attempt_id"), "artifact_metadata_sha256": metadata_hash,
            "source_artifact_path": metadata["path"],
            "source_task_path": metadata.get("source", {}).get("task_path")})

    all_ids = set()
    for result_id in run.get("result_ids", []):
        result, _ = read(store / "runs" / RUN_ID / "results" / (result_id + ".json"))
        all_ids.update(result.get("artifact_ids", []))
    # Read only metadata while selecting; unrelated binary Artifacts are never opened.
    metadata_rows = []
    for identifier in all_ids:
        metadata, _ = read(store / "artifacts" / identifier / "artifact.json")
        metadata_rows.append(metadata)
    scientific_parents = {item.get("parent_attempt_id") for item in run.get("internal_calls", [])
                          if item.get("started") is True}
    if len(scientific_parents) != 1:
        raise RuntimeError("Expected one original Python producer of the four ORCA computations")
    producer = None
    for attempt in reversed(run.get("attempts", [])):
        if (attempt.get("execution_kind") != "python" or attempt["state"] != "completed"
                or attempt["id"] not in scientific_parents):
            continue
        rows = [item for item in metadata_rows if item.get("attempt_id") == attempt["id"]]
        by_name = {}
        for item in rows:
            by_name.setdefault(Path(item["path"]).name, []).append(item)
        if set(CANDIDATES + TABLES) <= set(by_name):
            producer = attempt
            break
    if producer is None:
        raise RuntimeError("No completed Python Attempt retains all 12 candidates and required tables")
    for name in CANDIDATES + TABLES:
        choices = sorted(by_name[name], key=lambda item: (
            item.get("source", {}).get("task_path") != "task/" + name, item["id"]))
        if len({item["sha256"] for item in choices}) != 1:
            raise RuntimeError("Conflicting retained products with the same filename: " + name)
        add(("candidates/" if name in CANDIDATES else "results/") + name, choices[0]["id"],
            purpose="retained_model_generated_product")
    prepared = [item["prepared_geometry"] for item in run.get("input_bindings", {}).values()
                if "prepared_geometry" in item]
    if len(prepared) != 1:
        raise RuntimeError("Expected the single retained initial geometry binding")
    add("initial/initial.xyz", prepared[0]["artifact_id"], expected=prepared[0]["sha256"],
        purpose="unchanged_initial_geometry")
    developer_hashes = {sha(path.read_bytes()) for path in batch.glob("deterministic_metadata_repair*.py")
                        if path.is_file()}
    developer_code_ids, developer_attempts, developer_phases = [], set(), []
    for phase in phases:
        freeze, _ = read(phase / "freeze.json")
        if freeze.get("developer_assisted") is not True:
            continue
        developer_phases.append(phase.name)
        execution_path = phase / "metadata-execution.json"
        if execution_path.is_file():
            execution, _ = read(execution_path)
            result, _ = read(store / "runs" / RUN_ID / "results" / (execution["result_id"] + ".json"))
            developer_attempts.add(result["attempt_id"])
    for index, attempt in enumerate(run.get("attempts", []), 1):
        if attempt.get("execution_kind") != "python" or attempt.get("started") is not True:
            continue
        reference = attempt["frozen_step"]["parameters"]["code"]
        origin = ("developer_assisted_metadata" if attempt["id"] in developer_attempts
                  or reference["sha256"] in developer_hashes else "model_generated")
        if origin == "developer_assisted_metadata":
            developer_code_ids.append(reference["artifact_id"])
        add(f"code/attempt-{index:03}.py", reference["artifact_id"], expected=reference["sha256"],
            purpose="actually_started_" + origin + "_Python_" + attempt["state"])
        selected[-1]["code_origin"] = origin
    calls = [item for item in run.get("internal_calls", []) if item.get("started") is True]
    if len(calls) != 4 or any(item["state"] != "completed" for item in calls):
        raise RuntimeError("Expected the four completed actual ORCA receipts")
    for call in calls:
        key = call["request_key"]
        if not re.fullmatch(r"[A-Za-z0-9_-]{1,96}", key):
            raise RuntimeError("Invalid internal request key")
        files = call["response"]["files"]
        for source_name, target in (("stdout.out", "stdout.out"), ("job.inp", "executed.inp")):
            reference = files[source_name]
            add(f"orca/{key}/{target}", reference["artifact_id"], expected=reference["sha256"],
                purpose="actual_ORCA_" + source_name)
        reference = call["code"]
        add(f"orca/{key}/submitted.inp", reference["artifact_id"], expected=reference["sha256"],
            purpose="frozen_SDK_submitted_input")
    payloads["audit/independent-audit.json"] = audit_raw
    selected.append({"path": "audit/independent-audit.json", "purpose": "independent_read_only_audit",
        "artifact_id": None, "sha256": sha(audit_raw), "bytes": len(audit_raw),
        "source_path": str(audit_path.resolve())})
    manifest = {"schema_version": 1, "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "run_id": RUN_ID, "run_state": run["state"], "delivery_status": run.get("delivery_status"),
        "goal_status": run.get("goal_status"), "budget_revision": run.get("budget_revision"),
        "usage": run["usage"], "producer_attempt_id": producer["id"],
        "scientific_inventory_basis": "Original completed model Python parent of the four ORCA receipts; later metadata copies do not replace inventory or values.",
        "developer_assistance": {"executed_metadata_code_artifact_ids": developer_code_ids,
            "declared_phases": developer_phases,
            "entire_process_claimed_autonomous": False},
        "run_source_sha256": sha(run_raw), "outcome_source_sha256": sha(outcome_raw),
        "files": selected, "original_file_bytes_preserved": True,
        "scientific_values_recomputed_or_edited": False, "task_code_executed": False,
        "limitations": ["This package copies retained evidence; it does not itself establish scientific qualification.",
            "GBW and other binary or unrelated execution files remain outside this curated delivery."]}
    manifest_raw = encode(manifest)
    payloads["manifest.json"] = manifest_raw
    if any(len(raw) > MAX_FILE_BYTES for raw in payloads.values()) or sum(map(len, payloads.values())) > MAX_TOTAL_BYTES:
        raise RuntimeError("Curated text delivery exceeds its fixed package limit")
    receipt_path = output.with_suffix(".manifest.json")
    temporary = output.with_suffix(".zip.building")
    if any(path.exists() for path in (output, receipt_path, temporary)):
        raise RuntimeError("Refusing to overwrite an existing delivery or incomplete build")
    with ZipFile(temporary, "x", compression=ZIP_DEFLATED, compresslevel=9) as archive:
        for name, raw in sorted(payloads.items()):
            entry = ZipInfo(name, date_time=(1980, 1, 1, 0, 0, 0))
            entry.compress_type = ZIP_DEFLATED
            archive.writestr(entry, raw)
    with ZipFile(temporary) as archive:
        if set(archive.namelist()) != set(payloads):
            raise RuntimeError("ZIP inventory differs")
        for name, raw in payloads.items():
            if archive.read(name) != raw:
                raise RuntimeError("ZIP does not preserve source bytes: " + name)
    if run_path.read_bytes() != run_raw or outcome_path.read_bytes() != outcome_raw or audit_path.read_bytes() != audit_raw:
        raise RuntimeError("Final sources changed while packaging")
    for metadata, raw, path, _ in artifact_cache.values():
        if path.read_bytes() != raw or sha(raw) != metadata["sha256"]:
            raise RuntimeError("Artifact changed while packaging")
    os.link(temporary, output)
    temporary.unlink()
    with receipt_path.open("xb") as stream:
        stream.write(manifest_raw)
    return {"zip": str(output), "sha256": sha(output.read_bytes()), "bytes": output.stat().st_size,
            "entry_count": len(payloads), "manifest": str(receipt_path), "run_id": RUN_ID}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--finalize", action="store_true")
    parser.add_argument("--audit", type=Path, required=True)
    args = parser.parse_args()
    if not args.finalize:
        parser.error("Wait for final continuation and independent audit, then pass --finalize")
    batch = Path(__file__).resolve().parent
    print(json.dumps(build(batch, args.audit, batch / "hexane-results.zip"), ensure_ascii=False))


if __name__ == "__main__":
    main()
