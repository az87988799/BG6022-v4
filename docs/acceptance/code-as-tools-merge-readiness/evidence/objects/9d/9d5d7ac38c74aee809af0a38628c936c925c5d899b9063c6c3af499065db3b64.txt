import json, re
from pathlib import Path
from orca_task_sdk import run_orca, receipt_source
Path('water.inp').write_text('! HF STO-3G RHF SP TightSCF NoRI NoAutoStart\n%maxcore 192\n%pal nprocs 4 end\n%scf MaxIter 100 AutoTRAH false end\n* xyzfile 0 1 initial.xyz\n')
receipt = run_orca('water.inp', dependencies=['initial.xyz'], request_key='water')
Path('receipt.json').write_text(json.dumps(receipt))
assert receipt['operation_status'] == 'completed', receipt
text = Path(receipt['files']['stdout.out']).read_text()
assert '4 parallel MPI-processes' in text
energy = float(re.findall(r'FINAL SINGLE POINT ENERGY\s+(-?\d+\.\d+)', text)[-1])
Path('observations.json').write_text(json.dumps({'version': 1, 'observations': [{
    'name': 'energy', 'value': energy, 'unit': 'Eh',
    'sources': [receipt_source(receipt, 'stdout.out')],
    'limitations': ['Developer smoke; generated parsing is unverified.']
}]}))
